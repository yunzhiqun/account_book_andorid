### 1.2 DAO - AccountDao.kt

```kotlin
package com.accountbook.app.data.dao

import androidx.room.*
import com.accountbook.app.data.entity.AccountRecord
import kotlinx.coroutines.flow.Flow

@Dao
interface AccountDao {

    // 查询所有记录（按时间倒序）
    @Query("SELECT * FROM account_records ORDER BY createdAt DESC")
    fun getAllRecords(): Flow<List<AccountRecord>>

    // 按时间段查询
    @Query("""
        SELECT * FROM account_records 
        WHERE createdAt BETWEEN :startTime AND :endTime 
        ORDER BY createdAt DESC
    """)
    fun getRecordsByDateRange(startTime: Long, endTime: Long): Flow<List<AccountRecord>>

    // 按月份查询（用于默认展示）
    @Query("""
        SELECT * FROM account_records 
        WHERE createdAt BETWEEN :startTime AND :endTime 
        ORDER BY createdAt DESC
    """)
    fun getRecordsByMonth(startTime: Long, endTime: Long): Flow<List<AccountRecord>>

    // 插入单条记录
    @Insert(onConflict = OnConflictStrategy.REPLACE)
    suspend fun insertRecord(record: AccountRecord): Long

    // 更新记录
    @Update
    suspend fun updateRecord(record: AccountRecord)

    // 删除单条记录
    @Delete
    suspend fun deleteRecord(record: AccountRecord)

    // 批量删除
    @Query("DELETE FROM account_records WHERE id IN (:ids)")
    suspend fun deleteBatch(ids: List<Long>)

    // 根据 ID 查询单条记录
    @Query("SELECT * FROM account_records WHERE id = :id")
    suspend fun getRecordById(id: Long): AccountRecord?

    // 查询所有年份（用于折叠展示）
    @Query("""
        SELECT DISTINCT CAST(strftime('%Y', datetime(createdAt/1000, 'unixepoch', 'localtime')) AS INTEGER) 
        FROM account_records 
        ORDER BY 1 DESC
    """)
    fun getAllYears(): Flow<List<Int>>
}
```

```kotlin
package com.accountbook.app.viewmodel

import androidx.lifecycle.ViewModel
import androidx.lifecycle.viewModelScope
import com.accountbook.app.data.entity.AccountRecord
import com.accountbook.app.data.repository.AccountRepository
import com.accountbook.app.util.DateUtils
import dagger.hilt.android.lifecycle.HiltViewModel
import kotlinx.coroutines.flow.*
import kotlinx.coroutines.launch
import javax.inject.Inject

@HiltViewModel
class AccountViewModel @Inject constructor(
    private val repository: AccountRepository
) : ViewModel() {

    // 主页 UI 状态
    private val _uiState = MutableStateFlow(HomeUiState())
    val uiState: StateFlow<HomeUiState> = _uiState.asStateFlow()

    // 当前展示的记录（用于总计计算和复制）
    private val _currentRecords = MutableStateFlow<List<AccountRecord>>(emptyList())
    val currentRecords: StateFlow<List<AccountRecord>> = _currentRecords.asStateFlow()

    // 总计金额
    private val _totalAmount = MutableStateFlow(0.0)
    val totalAmount: StateFlow<Double> = _totalAmount.asStateFlow()

    // 搜索状态
    private val _searchUiState = MutableStateFlow(SearchUiState())
    val searchUiState: StateFlow<SearchUiState> = _searchUiState.asStateFlow()

    init {
        // 初始化：加载当前月份数据
        loadCurrentMonthData()
    }

    // 加载当前月份数据
    fun loadCurrentMonthData() {
        val (startTime, endTime) = DateUtils.getCurrentMonthRange()
        repository.getRecordsByMonth(startTime, endTime)
            .onEach { records ->
                _currentRecords.value = records
                _totalAmount.value = records.sumOf { it.amount }
                _uiState.update { it.copy(isLoading = false) }
            }
            .launchIn(viewModelScope)
    }

    // 按时间段查询
    fun queryByDateRange(startTime: Long, endTime: Long) {
        _uiState.update { it.copy(isLoading = true) }
        repository.getRecordsByDateRange(startTime, endTime)
            .onEach { records ->
                _currentRecords.value = records
                _totalAmount.value = records.sumOf { it.amount }
                _uiState.update { it.copy(isLoading = false) }
            }
            .launchIn(viewModelScope)
    }

    // 新增记录
    fun insertRecord(amount: Double, note: String) {
        viewModelScope.launch {
            val record = AccountRecord(amount = amount, note = note)
            repository.insertRecord(record)
            // 插入后自动刷新（Flow 会自动触发）
        }
    }

    // 更新记录
    fun updateRecord(record: AccountRecord) {
        viewModelScope.launch {
            repository.updateRecord(record)
        }
    }

    // 删除单条记录
    fun deleteRecord(record: AccountRecord) {
        viewModelScope.launch {
            repository.deleteRecord(record)
        }
    }

    // 批量删除
    fun deleteBatch(ids: List<Long>) {
        viewModelScope.launch {
            repository.deleteBatch(ids)
            // 退出选择模式
            _uiState.update { it.copy(isSelectionMode = false, selectedIds = emptySet()) }
        }
    }

    // 切换选择模式
    fun toggleSelectionMode() {
        _uiState.update { 
            it.copy(
                isSelectionMode = !it.isSelectionMode,
                selectedIds = if (it.isSelectionMode) emptySet() else it.selectedIds
            )
        }
    }

    // 切换记录选中状态
    fun toggleRecordSelection(recordId: Long) {
        _uiState.update { state ->
            val newSelectedIds = state.selectedIds.toMutableSet()
            if (newSelectedIds.contains(recordId)) {
                newSelectedIds.remove(recordId)
            } else {
                newSelectedIds.add(recordId)
            }
            state.copy(selectedIds = newSelectedIds)
        }
    }

    // 全选当前可见记录
    fun selectAllVisible() {
        _uiState.update { state ->
            state.copy(selectedIds = _currentRecords.value.map { it.id }.toSet())
        }
    }

    // 生成复制文本
    fun generateCopyText(): String {
        val amounts = _currentRecords.value
            .sortedBy { it.createdAt }
            .map { it.amount }
        val total = amounts.sum()
        val equation = amounts.joinToString("+") { amount ->
            if (amount == amount.toLong().toDouble()) {
                amount.toLong().toString()
            } else {
                amount.toString()
            }
        }
        val totalStr = if (total == total.toLong().toDouble()) {
            total.toLong().toString()
        } else {
            total.toString()
        }
        return "$equation=$totalStr"
    }
}

// UI 状态数据类
data class HomeUiState(
    val isLoading: Boolean = true,
    val isSelectionMode: Boolean = false,
    val selectedIds: Set<Long> = emptySet()
)

data class SearchUiState(
    val startYear: String = "",
    val startMonth: String = "",
    val startDay: String = "",
    val startHour: String = "",
    val startMinute: String = "",
    val startSecond: String = "",
    val endYear: String = "",
    val endMonth: String = "",
    val endDay: String = "",
    val endHour: String = "",
    val endMinute: String = "",
    val endSecond: String = ""
)
```

```kotlin
@Composable
fun AddEditDialog(
    record: AccountRecord? = null, // null 表示新增，非 null 表示编辑
    onDismiss: () -> Unit,
    onConfirm: (amount: Double, note: String) -> Unit
) {
    var amountText by remember { mutableStateOf(record?.amount?.toString() ?: "") }
    var noteText by remember { mutableStateOf(record?.note ?: "") }
    var amountError by remember { mutableStateOf<String?>(null) }

    AlertDialog(
        onDismissRequest = onDismiss,
        title = { Text(if (record == null) "新增账单" else "编辑账单") },
        text = {
            Column {
                // 金额输入（必填）
                OutlinedTextField(
                    value = amountText,
                    onValueChange = { value ->
                        // 限制只能输入数字和小数点
                        if (value.isEmpty() || value.matches(Regex("^\\d*\\.?\\d*$"))) {
                            amountText = value
                            amountError = null
                        }
                    },
                    label = { Text("金额 *") },
                    placeholder = { Text("请输入金额") },
                    isError = amountError != null,
                    supportingText = { amountError?.let { Text(it) } },
                    keyboardOptions = KeyboardOptions(keyboardType = KeyboardType.NumberDecimal),
                    singleLine = true,
                    modifier = Modifier.fillMaxWidth()
                )

                Spacer(modifier = Modifier.height(16.dp))

                // 备注输入（选填）
                OutlinedTextField(
                    value = noteText,
                    onValueChange = { 
                        if (it.length <= 200) noteText = it 
                    },
                    label = { Text("备注（选填）") },
                    placeholder = { Text("添加备注信息") },
                    maxLines = 3,
                    modifier = Modifier.fillMaxWidth()
                )
            }
        },
        confirmButton = {
            TextButton(
                onClick = {
                    val amount = amountText.toDoubleOrNull()
                    when {
                        amount == null -> amountError = "请输入有效金额"
                        amount < 0 -> amountError = "金额不能为负数"
                        else -> onConfirm(amount, noteText)
                    }
                }
            ) {
                Text("保存", color = BluePrimary)
            }
        },
        dismissButton = {
            TextButton(onClick = onDismiss) {
                Text("取消", color = TextSecondary)
            }
        }
    )
}
```

```kotlin
package com.accountbook.app.util

import java.util.Calendar

object DateUtils {
    
    // 获取当前月份的时间范围
    fun getCurrentMonthRange(): Pair<Long, Long> {
        val calendar = Calendar.getInstance()
        
        // 开始时间：当月1日 00:00:00
        calendar.set(Calendar.DAY_OF_MONTH, 1)
        calendar.set(Calendar.HOUR_OF_DAY, 0)
        calendar.set(Calendar.MINUTE, 0)
        calendar.set(Calendar.SECOND, 0)
        calendar.set(Calendar.MILLISECOND, 0)
        val startTime = calendar.timeInMillis
        
        // 结束时间：当月最后一天 23:59:59
        calendar.set(Calendar.DAY_OF_MONTH, calendar.getActualMaximum(Calendar.DAY_OF_MONTH))
        calendar.set(Calendar.HOUR_OF_DAY, 23)
        calendar.set(Calendar.MINUTE, 59)
        calendar.set(Calendar.SECOND, 59)
        calendar.set(Calendar.MILLISECOND, 999)
        val endTime = calendar.timeInMillis
        
        return Pair(startTime, endTime)
    }
    
    // 解析搜索范围（处理默认值）
    fun parseSearchRange(state: SearchUiState): Pair<Long, Long> {
        val calendar = Calendar.getInstance()
        val currentYear = calendar.get(Calendar.YEAR)
        val currentMonth = calendar.get(Calendar.MONTH) + 1
        val currentDay = calendar.get(Calendar.DAY_OF_MONTH)
        
        // 开始时间
        val startYear = state.startYear.toIntOrNull() ?: currentYear
        val startMonth = state.startMonth.toIntOrNull() ?: currentMonth
        val startDay = state.startDay.toIntOrNull() ?: currentDay
        val startHour = state.startHour.toIntOrNull() ?: 0
        val startMinute = state.startMinute.toIntOrNull() ?: 0
        val startSecond = state.startSecond.toIntOrNull() ?: 0
        
        calendar.set(startYear, startMonth - 1, startDay, startHour, startMinute, startSecond)
        calendar.set(Calendar.MILLISECOND, 0)
        val startTime = calendar.timeInMillis
        
        // 结束时间
        val endYear = state.endYear.toIntOrNull() ?: currentYear
        val endMonth = state.endMonth.toIntOrNull() ?: currentMonth
        val endDay = state.endDay.toIntOrNull() ?: currentDay
        val endHour = state.endHour.toIntOrNull() ?: 23
        val endMinute = state.endMinute.toIntOrNull() ?: 59
        val endSecond = state.endSecond.toIntOrNull() ?: 59
        
        calendar.set(endYear, endMonth - 1, endDay, endHour, endMinute, endSecond)
        calendar.set(Calendar.MILLISECOND, 999)
        val endTime = calendar.timeInMillis
        
        return Pair(startTime, endTime)
    }
    
    // 格式化日期时间
    fun formatDateTime(timestamp: Long): String {
        val calendar = Calendar.getInstance()
        calendar.timeInMillis = timestamp
        val year = calendar.get(Calendar.YEAR)
        val month = calendar.get(Calendar.MONTH) + 1
        val day = calendar.get(Calendar.DAY_OF_MONTH)
        val hour = calendar.get(Calendar.HOUR_OF_DAY)
        val minute = calendar.get(Calendar.MINUTE)
        
        return "$year-${month.toString().padStart(2, '0')}-${day.toString().padStart(2, '0')} " +
               "${hour.toString().padStart(2, '0')}:${minute.toString().padStart(2, '0')}"
    }
    
    // 从时间戳提取年份
    fun getYearFromTimestamp(timestamp: Long): Int {
        val calendar = Calendar.getInstance()
        calendar.timeInMillis = timestamp
        return calendar.get(Calendar.YEAR)
    }
    
    // 从时间戳提取月份
    fun getMonthFromTimestamp(timestamp: Long): Int {
        val calendar = Calendar.getInstance()
        calendar.timeInMillis = timestamp
        return calendar.get(Calendar.MONTH) + 1
    }
}
```

```kotlin
package com.accountbook.app.ui.navigation

import androidx.compose.runtime.Composable
import androidx.hilt.navigation.compose.hiltViewModel
import androidx.navigation.compose.NavHost

import androidx.navigation.compose.rememberNavController
import com.accountbook.app.ui.screens.HomeScreen
import com.accountbook.app.ui.screens.SearchScreen
import com.accountbook.app.viewmodel.AccountViewModel

@Composable
fun NavGraph() {
    val navController = rememberNavController()
    val viewModel: AccountViewModel = hiltViewModel()

    NavHost(navController = navController, startDestination = "home") {
        composable("home") {
            HomeScreen(
                viewModel = viewModel,
                onNavigateToSearch = { navController.navigate("search") }
            )
        }
        composable("search") {
            SearchScreen(
                viewModel = viewModel,
                onNavigateBack = { navController.popBackStack() }
            )
        }
    }
```
